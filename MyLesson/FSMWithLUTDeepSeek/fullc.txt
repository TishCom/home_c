#include "fsm.h"
#include <stdlib.h>
#include <string.h>

/*--------------------------------------------*
 * ВСПОМОГАТЕЛЬНЫЕ ФУНКЦИИ
 *--------------------------------------------*/

// Проверка существования состояния в таблице
static bool is_valid_state(const fsm_hierarchy_t *fsm, uint32_t state)
{
    if (fsm == NULL || fsm->table == NULL) return false;
    for (uint32_t i = 0; i < fsm->table_size; i++) {
        if (fsm->table[i].state == state) return true;
    }
    return false;
}

/*--------------------------------------------*
 * ОСНОВНЫЕ ФУНКЦИИ (ядро)
 *--------------------------------------------*/

void fsm_init(fsm_t *fsm, uint32_t initial_state,
              const fsm_transition_t *table, uint32_t table_size,
              fsm_state_handler_t default_handler)
{
    if (fsm == NULL) return;
    
    fsm->current_state = initial_state;
    fsm->table = table;
   fsm->table_size = table_size;
    fsm->default_handler = default_handler;
}

bool fsm_dispatch(fsm_t *fsm, uint32_t event)
{
    if (fsm == NULL || fsm->table == NULL) return false;
    
    uint32_t old_state = fsm->current_state;
    uint32_t new_state = old_state;
    bool handled = false;
    
    // Поиск подходящего перехода в таблице
    for (uint32_t i = 0; i < fsm->table_size; i++) {
        if (fsm->table[i].state == old_state && fsm->table[i].event == event) {
            // Вызываем обработчик, если он задан
            if (fsm->table[i].handler != NULL) {
                new_state = fsm->table[i].handler();
            } else {
                new_state = fsm->table[i].next_state;
            }
            handled = true;
            break;
        }
    }
    
    // Если переход не найден, используем обработчик по умолчанию
    if (!handled && fsm->default_handler != NULL) {
        new_state = fsm->default_handler();
        handled = true;
    }
    
    fsm->current_state = new_state;
    return handled;
}

uint32_t fsm_get_state(const fsm_t *fsm)
{
    return (fsm == NULL) ? 0 : fsm->current_state;
}

void fsm_set_state(fsm_t *fsm, uint32_t state)
{
    if (fsm != NULL) fsm->current_state = state;
}

void fsm_reset(fsm_t *fsm)
{
    if (fsm == NULL || fsm->table == NULL) return;
    fsm->current_state = fsm->table[0].state;
}

/*--------------------------------------------*
 * ИЕРАРХИЧЕСКИЕ ФУНКЦИИ
 *--------------------------------------------*/

void fsm_hierarchy_init(fsm_hierarchy_t *fsm, uint32_t initial_state,
                        const fsm_hierarchy_rule_t *table, uint32_t table_size,
                        fsm_state_handler_t default_handler)
{
    if (fsm == NULL) return;
    
    fsm->current_state = initial_state;
    fsm->table = table;
    fsm->table_size = table_size;
    fsm->default_handler = default_handler;
    fsm->initialized = true;
    
    // При инициализации вызываем on_enter для начального состояния
    for (uint32_t i = 0; i < table_size; i++) {
        if (table[i].state == initial_state && table[i].on_enter != NULL) {
            table[i].on_enter();
            break;
        }
    }
}

bool fsm_hierarchy_dispatch(fsm_hierarchy_t *fsm, uint32_t event)
{
    if (fsm == NULL || fsm->table == NULL) return false;
    
    uint32_t old_state = fsm->current_state;
    uint32_t new_state = old_state;
    bool handled = false;
    
    // Поиск перехода
    for (uint32_t i = 0; i < fsm->table_size; i++) {
        if (fsm->table[i].state == old_state && fsm->table[i].event == event) {
            if (fsm->table[i].handler != NULL) {
                new_state = fsm->table[i].handler();
            } else {
                new_state = fsm->table[i].next_state;
            }
            handled = true;
            break;
        }
    }
    
    // Обработчик по умолчанию
    if (!handled && fsm->default_handler != NULL) {
        new_state = fsm->default_handler();
        handled = true;
    }
    
    if (!handled) return false;
    
    // При смене состояния вызываем on_exit старого и on_enter нового
    if (new_state != old_state) {
        // Выход из старого состояния
        for (uint32_t i = 0; i < fsm->table_size; i++) {
            if (fsm->table[i].state == old_state && fsm->table[i].on_exit != NULL) {
                fsm->table[i].on_exit();
                break;
            }
        }
        
        fsm->current_state = new_state;
        
        // Вход в новое состояние
        for (uint32_t i = 0; i < fsm->table_size; i++) {
            if (fsm->table[i].state == new_state && fsm->table[i].on_enter != NULL) {
                fsm->table[i].on_enter();
                break;
            }
        }
    } else {
        fsm->current_state = new_state;
    }
    
    return true;
}

uint32_t fsm_hierarchy_get_state(const fsm_hierarchy_t *fsm)
{
    return (fsm == NULL) ? 0 : fsm->current_state;
}

void fsm_hierarchy_set_state(fsm_hierarchy_t *fsm, uint32_t state)
{
    if (fsm != NULL && is_valid_state(fsm, state)) {
        fsm->current_state = state;
    }
}

void fsm_hierarchy_reset(fsm_hierarchy_t *fsm)
{
    if (fsm == NULL || fsm->table == NULL) return;
    fsm->current_state = fsm->table[0].state;
}

/*--------------------------------------------*
 * ФУНКЦИИ ДЛЯ ВЫЗЫВАЕМЫХ АВТОМАТОВ
 *--------------------------------------------*/

void submachine_init(submachine_context_t *ctx, 
                     fsm_hierarchy_t *parent, fsm_hierarchy_t *child,
                     uint32_t parent_suspend_state, 
                     uint32_t parent_resume_event,
                     uint32_t child_finish_state)
{
    if (ctx == NULL) return;
    
    ctx->parent = parent;
    ctx->child = child;
    ctx->parent_suspend_state = parent_suspend_state;
    ctx->parent_resume_event = parent_resume_event;
    ctx->child_finish_state = child_finish_state;
    ctx->active = false;
    ctx->completed = false;
}

/* ВЫЗЫВАЕМЫЙ АВТОМАТ (блокирующий)
 * Работает как подпрограмма: выполняется до завершения,
 * после чего возвращает управление родителю
 */
bool submachine_call(submachine_context_t *ctx, uint32_t entry_event)
{
    if (ctx == NULL || ctx->parent == NULL || ctx->child == NULL) return false;
    
    // Проверяем, что родитель находится в состоянии ожидания
    if (fsm_hierarchy_get_state(ctx->parent) != ctx->parent_suspend_state) {
        return false;
    }
    
    // Сбрасываем дочерний автомат в начальное состояние
    fsm_hierarchy_reset(ctx->child);
    ctx->active = true;
    ctx->completed = false;
    
    // Обрабатываем входное событие, если оно передано
    if (entry_event != 0) {
        fsm_hierarchy_dispatch(ctx->child, entry_event);
    }
    
    // В реальной системе здесь должен быть цикл обработки событий.
    // Для простоты предполагаем, что дочерний автомат сам управляет
    // своими переходами через внутренние механизмы (таймеры, прерывания)
    // и по достижении конечного состояния вызывает parent_resume_event.
    // В данной реализации флаг завершения устанавливается вручную.
    
    ctx->active = false;
    ctx->completed = true;
    fsm_hierarchy_dispatch(ctx->parent, ctx->parent_resume_event);
    
    return true;
}

/*--------------------------------------------*
 * ВЛОЖЕННЫЕ АВТОМАТЫ (пошаговые)
 *--------------------------------------------*/

/* Запуск вложенного автомата
 * Только инициализирует, не выполняет
 */
void submachine_start(submachine_context_t *ctx, uint32_t entry_event)
{
    if (ctx == NULL || ctx->parent == NULL || ctx->child == NULL) return;
    
    // Проверяем, что родитель в состоянии ожидания
    if (fsm_hierarchy_get_state(ctx->parent) != ctx->parent_suspend_state) return;
    
    fsm_hierarchy_reset(ctx->child);
    if (entry_event != 0) {
        fsm_hierarchy_dispatch(ctx->child, entry_event);
    }
    ctx->active = true;
    ctx->completed = false;
}

/* Один шаг выполнения вложенного автомата
 * Вызывается в главном цикле или по таймеру
 */
bool submachine_step(submachine_context_t *ctx, uint32_t event)
{
    if (ctx == NULL || !ctx->active) return false;
    
    fsm_hierarchy_dispatch(ctx->child, event);
    
    // Проверяем достижение конечного состояния
    if (fsm_hierarchy_get_state(ctx->child) == ctx->child_finish_state) {
        ctx->active = false;
        ctx->completed = true;
        fsm_hierarchy_dispatch(ctx->parent, ctx->parent_resume_event);
        return true;  // автомат завершён
    }
    
    return false;  // ещё работает
}

bool submachine_is_active(const submachine_context_t *ctx)
{
    return (ctx != NULL) && ctx->active;
}

bool submachine_is_completed(const submachine_context_t *ctx)
{
    return (ctx != NULL) && ctx->completed;
}

void submachine_abort(submachine_context_t *ctx)
{
    if (ctx == NULL) return;
    ctx->active = false;
    ctx->completed = false;
    fsm_hierarchy_dispatch(ctx->parent, ctx->parent_resume_event);
}

void submachine_reset(submachine_context_t *ctx)
{
    if (ctx == NULL) return;
    ctx->active = false;
    ctx->completed = false;
    fsm_hierarchy_reset(ctx->child);
}

/*--------------------------------------------*
 * ФУНКЦИИ ДЛЯ ГРУППЫ АВТОМАТОВ
 *--------------------------------------------*/

void fsm_group_init(fsm_group_t *group, fsm_hierarchy_t **machines, uint32_t count)
{
    if (group == NULL) return;
    group->machines = machines;
    group->count = count;
}

void fsm_group_dispatch(fsm_group_t *group, uint32_t event)
{
    if (group == NULL || group->machines == NULL) return;
    
    // Отправляем событие всем автоматам в группе
    for (uint32_t i = 0; i < group->count; i++) {
        if (group->machines[i] != NULL) {
            fsm_hierarchy_dispatch(group->machines[i], event);
        }
    }
}

uint32_t fsm_group_count(const fsm_group_t *group)
{
    return (group == NULL) ? 0 : group->count;
}
