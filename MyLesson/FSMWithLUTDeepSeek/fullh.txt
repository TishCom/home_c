#ifndef FSM_H_
#define FSM_H_

#include <stdbool.h>
#include <inttypes.h>

/*--------------------------------------------*
 * ОСНОВНЫЕ ТИПЫ ДАННЫХ (ядро)
 *--------------------------------------------*/

// Указатель на функцию-обработчик состояния
// Возвращает следующее состояние автомата
typedef uint32_t (*fsm_state_handler_t)(void);

// Правило перехода для простого (неиерархического) автомата
typedef struct {
    uint32_t state;                     // текущее состояние
    uint32_t event;                     // событие, вызывающее переход
    fsm_state_handler_t handler;        // обработчик (если NULL, переход без вызова)
    uint32_t next_state;                // следующее состояние после перехода
} fsm_transition_t;

// Структура простого конечного автомата
typedef struct {
    uint32_t current_state;                         // текущее состояние
    const fsm_transition_t *table;                  // таблица переходов
    uint32_t table_size;                            // размер таблицы
    fsm_state_handler_t default_handler;            // обработчик для непредусмотренных событий
} fsm_t;

/*--------------------------------------------*
 * ИЕРАРХИЧЕСКИЕ СОСТОЯНИЯ (с on_enter/on_exit)
 *--------------------------------------------*/

// Правило перехода с обработчиками входа/выхода
typedef struct {
    uint32_t state;                     // текущее состояние
    uint32_t event;                     // событие, вызывающее переход
    fsm_state_handler_t handler;        // обработчик события (может быть NULL)
    uint32_t next_state;                // следующее состояние
    fsm_state_handler_t on_enter;       // вызывается при входе в состояние (может быть NULL)
    fsm_state_handler_t on_exit;        // вызывается при выходе из состояния (может быть NULL)
} fsm_hierarchy_rule_t;

// Структура иерархического автомата (с поддержкой on_enter/on_exit)
typedef struct {
    uint32_t current_state;                         // текущее состояние
    const fsm_hierarchy_rule_t *table;              // таблица переходов
    uint32_t table_size;                            // размер таблицы
    fsm_state_handler_t default_handler;            // обработчик для непредусмотренных событий
    bool initialized;                               // флаг инициализации
} fsm_hierarchy_t;

/*--------------------------------------------*
 * ВЫЗЫВАЕМЫЕ АВТОМАТЫ (суб-машины)
 * 
 * Вызываемый автомат подобен подпрограмме:
 * - При вызове начинает работу с начального состояния
 * - Работает до достижения конечного состояния
 * - Управление возвращается родителю только после завершения
 *--------------------------------------------*/

// Структура контекста для управления вызываемым автоматом
typedef struct {
    fsm_hierarchy_t *parent;            // указатель на родительский автомат
    fsm_hierarchy_t *child;             // указатель на дочерний автомат
    uint32_t parent_suspend_state;      // состояние родителя, в котором он ждёт завершения дочернего
    uint32_t parent_resume_event;       // событие, отправляемое родителю после завершения дочернего
    uint32_t child_finish_state;        // конечное состояние дочернего автомата (после достижения — выход)
    bool active;                        // активен ли в данный момент дочерний автомат
    bool completed;                     // флаг завершения дочернего автомата
} submachine_context_t;

/*--------------------------------------------*
 * ВЛОЖЕННЫЕ АВТОМАТЫ (параллельные)
 * 
 * Вложенные автоматы работают пошагово, параллельно с родителем:
 * - Каждый вызов делает один шаг (один такт работы)
 * - Состояние сохраняется между вызовами
 * - Не блокируют выполнение родителя
 *--------------------------------------------*/

// Группа автоматов для параллельного выполнения
typedef struct {
    fsm_hierarchy_t **machines;         // массив указателей на автоматы
    uint32_t count;                     // количество автоматов в группе
} fsm_group_t;

/*--------------------------------------------*
 * ОСНОВНЫЕ ФУНКЦИИ (ядро)
 *--------------------------------------------*/

/* Инициализация простого конечного автомата
 * Параметры:
 *   fsm           - указатель на структуру автомата
 *   initial_state - начальное состояние
 *   table         - таблица переходов
 *   table_size    - размер таблицы
 *   default_handler - обработчик по умолчанию (для непредусмотренных событий)
 */
void fsm_init(fsm_t *fsm, uint32_t initial_state,
              const fsm_transition_t *table, uint32_t table_size,
              fsm_state_handler_t default_handler);

/* Обработка события
 * Параметры:
 *   fsm   - указатель на автомат
 *   event - событие для обработки
 * Возвращает:
 *   true  - переход найден и обработан
 *   false - переход не найден
 */
bool fsm_dispatch(fsm_t *fsm, uint32_t event);

/* Получение текущего состояния
 * Параметры:
 *   fsm - указатель на автомат
 * Возвращает: текущее состояние (0 если fsm == NULL)
 */
uint32_t fsm_get_state(const fsm_t *fsm);

/* Принудительная установка состояния
 * Параметры:
 *   fsm   - указатель на автомат
 *   state - новое состояние
 */
void fsm_set_state(fsm_t *fsm, uint32_t state);

/* Сброс автомата в начальное состояние
 * Параметры:
 *   fsm - указатель на автомат
 */
void fsm_reset(fsm_t *fsm);

/*--------------------------------------------*
 * ИЕРАРХИЧЕСКИЕ ФУНКЦИИ (с on_enter/on_exit)
 *--------------------------------------------*/

/* Инициализация иерархического автомата
 * Параметры:
 *   fsm           - указатель на структуру автомата
 *   initial_state - начальное состояние
 *   table         - таблица переходов (с on_enter/on_exit)
 *   table_size    - размер таблицы
 *   default_handler - обработчик по умолчанию
 */
void fsm_hierarchy_init(fsm_hierarchy_t *fsm, uint32_t initial_state,
                        const fsm_hierarchy_rule_t *table, uint32_t table_size,
                        fsm_state_handler_t default_handler);

/* Обработка события иерархическим автоматом
 * Вызывает on_exit при выходе из состояния, on_enter при входе
 * Параметры:
 *   fsm   - указатель на автомат
 *   event - событие для обработки
 * Возвращает: true если переход обработан
 */
bool fsm_hierarchy_dispatch(fsm_hierarchy_t *fsm, uint32_t event);

/* Получение текущего состояния иерархического автомата
 * Параметры:
 *   fsm - указатель на автомат
 */
uint32_t fsm_hierarchy_get_state(const fsm_hierarchy_t *fsm);

/* Принудительная установка состояния иерархического автомата
 * Параметры:
 *   fsm   - указатель на автомат
 *   state - новое состояние
 */
void fsm_hierarchy_set_state(fsm_hierarchy_t *fsm, uint32_t state);

/* Сброс иерархического автомата в начальное состояние
 * Параметры:
 *   fsm - указатель на автомат
 */
void fsm_hierarchy_reset(fsm_hierarchy_t *fsm);

/*--------------------------------------------*
 * ФУНКЦИИ ДЛЯ ВЫЗЫВАЕМЫХ АВТОМАТОВ
 *--------------------------------------------*/

/* Инициализация контекста вызываемого автомата
 * Параметры:
 *   ctx                 - указатель на контекст
 *   parent              - родительский автомат
 *   child               - дочерний (вызываемый) автомат
 *   parent_suspend_state - состояние родителя, в котором он ожидает
 *   parent_resume_event  - событие для возобновления родителя после завершения
 *   child_finish_state   - конечное состояние дочернего автомата
 */
void submachine_init(submachine_context_t *ctx, 
                     fsm_hierarchy_t *parent, fsm_hierarchy_t *child,
                     uint32_t parent_suspend_state, 
                     uint32_t parent_resume_event,
                     uint32_t child_finish_state);

/* ВЫЗЫВАЕМЫЙ АВТОМАТ (блокирующий режим)
 * Вызываемый автомат подобен подпрограмме:
 * - Запускается с начального состояния
 * - Обрабатывает переданное событие (entry_event)
 * - Выполняется до достижения child_finish_state
 * - Только после этого управление возвращается родителю
 * 
 * Параметры:
 *   ctx         - контекст вызываемого автомата
 *   entry_event - событие, передаваемое при вызове (первое событие)
 * Возвращает: true если вызов успешен
 */
bool submachine_call(submachine_context_t *ctx, uint32_t entry_event);

/*--------------------------------------------*
 * ФУНКЦИИ ДЛЯ ВЛОЖЕННЫХ АВТОМАТОВ (пошаговые)
 *--------------------------------------------*/

/* Запуск вложенного автомата (без выполнения)
 * Параметры:
 *   ctx         - контекст
 *   entry_event - начальное событие
 */
void submachine_start(submachine_context_t *ctx, uint32_t entry_event);

/* Один шаг выполнения вложенного автомата
 * Параметры:
 *   ctx   - контекст
 *   event - событие для обработки (может быть получено из очереди)
 * Возвращает: true если автомат завершён
 */
bool submachine_step(submachine_context_t *ctx, uint32_t event);

/* Проверка, активен ли вложенный автомат
 * Параметры:
 *   ctx - контекст
 * Возвращает: true если автомат в процессе выполнения
 */
bool submachine_is_active(const submachine_context_t *ctx);

/* Проверка, завершён ли вложенный автомат
 * Параметры:
 *   ctx - контекст
 * Возвращает: true если автомат успешно завершён
 */
bool submachine_is_completed(const submachine_context_t *ctx);

/* Принудительное завершение вложенного автомата
 * Параметры:
 *   ctx - контекст
 */
void submachine_abort(submachine_context_t *ctx);

/* Сброс вложенного автомата
 * Параметры:
 *   ctx - контекст
 */
void submachine_reset(submachine_context_t *ctx);

/*--------------------------------------------*
 * ФУНКЦИИ ДЛЯ ГРУППЫ АВТОМАТОВ (параллельные)
 *--------------------------------------------*/

/* Инициализация группы для параллельного выполнения
 * Параметры:
 *   group    - указатель на группу
 *   machines - массив указателей на автоматы
 *   count    - количество автоматов в массиве
 */
void fsm_group_init(fsm_group_t *group, fsm_hierarchy_t **machines, uint32_t count);

/* Отправка события всем автоматам в группе
 * Параметры:
 *   group - указатель на группу
 *   event - событие
 */
void fsm_group_dispatch(fsm_group_t *group, uint32_t event);

/* Получение количества автоматов в группе
 * Параметры:
 *   group - указатель на группу
 */
uint32_t fsm_group_count(const fsm_group_t *group);

#endif /* FSM_H_ */
